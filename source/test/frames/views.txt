# The reference frames scripts/capture.sh compares with, one view a line: its name (the reference
# is <name>.png beside this file), then render_test's arguments, with $CONTENT the test content as
# the build cooks it, $PWD the repository and a | for a space within one, as in a console line given
# to --run. The scenes are the test content's (source/test/content/scene), static: what the retired
# space sample built, baked into the files.
#
# The orbit scene as it opens: Ostra Highport over Ostra, the planet's limb behind it, the hull
# lit and reflecting the planet through the environment's capture.
orbit --scene $CONTENT/scene/orbit.yaml
# The same view drawn offscreen (--offscreen), with no surface or swapchain: its reference is a copy
# of orbit.png, so the offscreen target is held to the swapchain's frame to the byte.
orbit-offscreen --scene $CONTENT/scene/orbit.yaml --offscreen
# Ostra whole from 20,000 km, by day: its air's blue limb, the haze thickening towards it, clouds.
ostra --scene $CONTENT/scene/orbit.yaml --run orbit|0.8|-0.1|2.0e7|ostra
# Its night side from 14,000 km: the cities' lights, and the limb lit from behind.
ostra-night --scene $CONTENT/scene/orbit.yaml --run orbit|3.9|-0.1|1.4e7|ostra
# Corva from 450,000 km: its rings' three bands and two gaps, its shadow on them, and theirs on it.
corva --scene $CONTENT/scene/vesper.yaml --run orbit|0.8|-0.35|4.5e8|corva
# The Vesper scene as it opens: Corva Refinery with the rings below it and Corva's limb behind.
refinery --scene $CONTENT/scene/vesper.yaml
# The transparent pass's test bench three seconds in: a sphere of clear glass and a pane of amber
# glass over the stripes, an engine's plume, smoke lit by the sun and the lamp, and sparks.
effects --scene $CONTENT/scene/effects.yaml --after-frames 180
# The post chain's test bench two seconds in: occlusion where the crates meet the deck, the selected
# crate's amber line and the targeted pod's red glow, two lamp panels, the right one left alone by
# bloom, and the beacon at the bright half of its blink, lighting the wall below it.
post --scene $CONTENT/scene/post.yaml --after-frames 120 --ambient-occlusion
# The orbit scene as it opens, graded by the cold-light table: the shadows towards teal and the lit
# planet towards amber, a little more contrast and a little less colour than orbit.png.
graded --scene $CONTENT/scene/orbit.yaml --run grade|cold-light
# The docking sequence 1.5 s in, its roles the scene's ship and pad, from an orbit camera 30 m off
# the shuttle, where the chase camera was: swinging round its starboard side, three quarters through
# the blend to the first shot (an ease in and out), the clamps rising, and the letterbox's bars as
# far drawn.
dock-sequence --scene $CONTENT/scene/orbit.yaml --run orbit|3.14|0.15|30|ship --run sequence|dock|1
# The texture table's bench: the sixteen-slot fixture program on a board, each slot in a cell of a four
# by four grid from the bottom left, seven textures past the eighth slot and four fallbacks among them.
slots --scene $CONTENT/scene/slots.yaml
# The models scene as it opens (the smoke test's default): the cooked crates and the skinned arm on
# the deck, each model's base colour a KTX 2 image the cook encoded from the source's PNG.
models --scene $CONTENT/scene/main.yaml
# The generated asteroid (scene/asteroid.yaml) from 88 m, 165 m and 400 m off its centre: drawn at
# its second, fourth and fifth levels of detail (ModelTests.cpp's asteroid case checks each, and that
# a flight in and out across the switches changes the level once each way).
asteroid-near --scene $CONTENT/scene/asteroid.yaml --run orbit|0.6|0.2|88|asteroid
asteroid-mid --scene $CONTENT/scene/asteroid.yaml --run orbit|0.6|0.2|165|asteroid
asteroid-far --scene $CONTENT/scene/asteroid.yaml --run orbit|0.6|0.2|400|asteroid
# The ground sample as it opens (--ground: its own program, over its own content): the rover on its
# slab, from the chase camera behind it.
ground --ground
# The orbit scene with the engine's saves screen shown in German, every word of it from the engine's
# tables (assets/strings/de): its title, the empty list's line and the footer.
saves-de --scene $CONTENT/scene/orbit.yaml --run language|de --run saves|on
